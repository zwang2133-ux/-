<script>
    let records = JSON.parse(localStorage.getItem('my_records') || '[]');

    function switchPage(pageId) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
        
        document.getElementById('page-' + pageId).classList.add('active');
        const navBtn = document.getElementById('nav-' + pageId);
        if(navBtn) navBtn.classList.add('active');
    }

    function toggleDrawer() {
        document.getElementById('drawer').classList.toggle('open');
    }

    function setTheme(color) {
        document.documentElement.style.setProperty('--primary', color);
        localStorage.setItem('app_theme', color);
    }

    const savedTheme = localStorage.getItem('app_theme');
    if(savedTheme) setTheme(savedTheme);

    function selectTag(element, type) {
        element.parentElement.querySelectorAll('.tag').forEach(t => t.classList.remove('selected'));
        element.classList.add('selected');
    }

    function saveRecord() {
        const duration = document.getElementById('input-duration').value;
        const form = document.querySelector('#form-tags .tag.selected').innerText;
        const tool = document.querySelector('#tool-tags .tag.selected').innerText;
        
        const newRecord = {
            date: new Date().toISOString(),
            duration: parseInt(duration),
            form,
            tool
        };

        records.push(newRecord);
        localStorage.setItem('my_records', JSON.stringify(records));
        
        alert('記錄成功！');
        updateStats();
        switchPage('overview');
    }

    function updateStats() {
        document.getElementById('stat-count').innerText = records.length + ' 次';
        if(records.length > 0) {
            const totalMin = records.reduce((acc, curr) => acc + curr.duration, 0);
            document.getElementById('stat-duration').innerText = totalMin + ' 分鐘';
        }
    }

    const now = new Date();
    document.getElementById('current-date-str').innerText = `${now.getFullYear()}年${now.getMonth()+1}月${now.getDate()}日`;
    
    updateStats();
</script>
